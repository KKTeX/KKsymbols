% Regression test for loading KKran before KKsymbols.
% Compile with: latexmk -r .latexmkrc test-kkran-order.tex
\documentclass{jlreq}
\usepackage{KKran}
\usepackage{KKsymbols}

\makeatletter
\newcommand{\KKSordersmall}[1]{\def\KKSorderactual{SMALL}\typeout{KKS-ORDER-ROUTE:SMALL}}
\newcommand{\KKSordernormal}[1]{\def\KKSorderactual{NORMAL}\typeout{KKS-ORDER-ROUTE:NORMAL}}
\newcommand{\KKSorderexpect}[2]{%
  \def\KKSorderactual{UNSET}%
  \kks@autolowercase@select{\KKSordersmall}{\KKSordernormal}{#2}%
  \def\KKSorderexpected{#1}%
  \ifx\KKSorderactual\KKSorderexpected\else
    \PackageError{test-kkran-order}{Expected #1 route, got \KKSorderactual}{}%
  \fi}
\KKSorderexpect{NORMAL}{\Rrnum{8}}% multi-letter
\KKSorderexpect{SMALL}{\Rrnum{1}}% single-letter result
\KKSorderexpect{NORMAL}{\Rrnum{2-1}}% expands exactly as KKran: ii-1
\KKSorderexpect{NORMAL}{\Rrnum*{8}}%
\KKSorderexpect{SMALL}{\alphabetarray{2}}%
\KKSorderexpect{SMALL}{\alphabetarray{-1}}% KKran indexes from the end
\KKSorderexpect{NORMAL}{\alphabetarray*{2}}%
\KKSorderexpect{NORMAL}{\alphabetarray{27}}% empty in KKran
\makeatother

\begin{document}
\fbox{\maru{\Rrnum{8}}}\fbox{\maru*{\Rrnum{8}}}\quad
\fbox{\maru{\alphabetarray{2}}}\fbox{\maru*{\alphabetarray{2}}}
\end{document}
